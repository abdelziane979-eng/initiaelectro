// ========================================== // LOGIQUE DE VÉRIFICATION DU CÂBLAGE // ========================================== function verifierCablage() { const msgBox = document.getElementById('message'); msgBox.className = 'message'; msgBox.style.display = 'none'; // 1. Vérifier si les composants minimum sont présents const nbBatteries = composants.filter(c => c.type === 'batterie').length; const nbPDB = composants.filter(c => c.type === 'pdb').length; const nbFC = composants.filter(c => c.type === 'fc').length; const nbESC = composants.filter(c => c.type === 'esc').length; const nbMoteurs = composants.filter(c => c.type === 'moteur').length; if (nbBatteries < 1 || nbPDB < 1 || nbFC < 1 || nbESC < 4 || nbMoteurs < 4) { afficherMessage("❌ Il manque des composants ! Place 1 Batterie, 1 PDB, 1 Contrôleur de Vol, 4 ESC et 4 Moteurs.", "erreur"); return; } let erreurs = []; // Fonction utilitaire pour vérifier si deux pattes sont reliées function estConnecte(typeA, patteA, typeB, patteB) { return fils.some(f => { const cA = composants.find(c => c.id === f.compA); const cB = composants.find(c => c.id === f.compB); if (!cA || !cB) return false; // On vérifie dans les deux sens (A vers B ou B vers A) return ( (cA.type === typeA && f.patteA === patteA && cB.type === typeB && f.patteB === patteB) || (cB.type === typeA && f.patteB === patteA && cA.type === typeB && f.patteA === patteB) ); }); } // 2. Vérifier l'alimentation principale (Batterie -> PDB) if (!estConnecte('batterie', '+', 'pdb', 'bat+')) erreurs.push("La borne + de la Batterie n'est pas reliée au BAT+ du PDB."); if (!estConnecte('batterie', '-', 'pdb', 'bat-')) erreurs.push("La borne - de la Batterie n'est pas reliée au BAT- du PDB."); // 3. Vérifier l'alimentation du Contrôleur de Vol (PDB -> FC) if (!estConnecte('pdb', '5v', 'fc', '5v')) erreurs.push("Le 5V BEC du PDB n'est pas relié au 5V du FC."); if (!estConnecte('pdb', 'bat-', 'fc', 'gnd')) erreurs.push("La masse (GND) du FC n'est pas reliée au PDB."); // 4. Vérifier les ESC (Alimentation et Signal) // Pour simplifier, on vérifie qu'il y a au moins un câblage correct pour chaque patte du FC let m1OK = false, m2OK = false, m3OK = false, m4OK = false; composants.filter(c => c.type === 'esc').forEach(esc => { // Vérifier l'alimentation de l'ESC (PDB -> ESC) const escAlimente = fils.some(f => { const cA = composants.find(c => c.id === f.compA); const cB = composants.find(c => c.id === f.compB); return (cA.type === 'pdb' && cA.id === composants.find(comp => comp.type === 'pdb').id && f.patteA.startsWith('esc') && cB.id === esc.id && f.patteB === 'pwr+') || (cB.type === 'pdb' && cB.id === composants.find(comp => comp.type === 'pdb').id && f.patteB.startsWith('esc') && cA.id === esc.id && f.patteA === 'pwr+'); }); if (!escAlimente) erreurs.push("Un ESC n'est pas alimenté par le PDB (pwr+)."); // Vérifier la masse de l'ESC (PDB bat- -> ESC pwr-) const escMasse = fils.some(f => { const cA = composants.find(c => c.id === f.compA); const cB = composants.find(c => c.id === f.compB); return (cA.type === 'pdb' && f.patteA === 'bat-' && cB.id === esc.id && f.patteB === 'pwr-') || (cB.type === 'pdb' && f.patteB === 'bat-' && cA.id === esc.id && f.patteA === 'pwr-'); }); if (!escMasse) erreurs.push("Un ESC n'est pas relié à la masse du PDB (pwr-)."); // Vérifier le signal (FC -> ESC) if (estConnecte('fc', 'm1', 'esc', 'sig')) m1OK = true; if (estConnecte('fc', 'm2', 'esc', 'sig')) m2OK = true; if (estConnecte('fc', 'm3', 'esc', 'sig')) m3OK = true; if (estConnecte('fc', 'm4', 'esc', 'sig')) m4OK = true; // Vérifier le moteur (ESC -> Moteur) if (!estConnecte('esc', 'mot+', 'moteur', '+') || !estConnecte('esc', 'mot-', 'moteur', '-')) { erreurs.push("Un ESC n'est pas correctement relié à son moteur (M+ et M-)."); } }); if (!m1OK) erreurs.push("Le signal du Moteur 1 (FC) n'est pas relié à un ESC (SIG)."); if (!m2OK) erreurs.push("Le signal du Moteur 2 (FC) n'est pas relié à un ESC (SIG)."); if (!m3OK) erreurs.push("Le signal du Moteur 3 (FC) n'est pas relié à un ESC (SIG)."); if (!m4OK) erreurs.push("Le signal du Moteur 4 (FC) n'est pas relié à un ESC (SIG)."); // Affichage du résultat if (erreurs.length > 0) { // On enlève les doublons d'erreurs (si 4 ESC ne sont pas alimentés, on ne veut pas 4 fois la même ligne) const erreursUniques = [...new Set(erreurs)]; afficherMessage("❌ Câblage incomplet :

• " + erreursUniques.join("
• "), "erreur"); } else { afficherMessage("🎉 FÉLICITATIONS !
Le câblage est parfait. Le drone est prêt pour la checklist de vol ! 🚁", "succes"); lancerConfettis(); } } function afficherMessage(texte, type) { const msgBox = document.getElementById('message'); msgBox.innerHTML = texte; msgBox.className = 'message ' + type; msgBox.style.display = 'block'; } // ========================================== // ANIMATION CONFETTIS (Pour la réussite) // ========================================== function lancerConfettis() { const couleurs = ['#fbbf24', '#22c55e', '#ef4444', '#3b82f6', '#a855f7']; for (let i = 0; i < 50; i++) { setTimeout(() => { const c = document.createElement('div'); c.style.cssText = `position:fixed; width:10px; height:10px; left:${Math.random()*100}%; top:-20px; background:${couleurs[Math.floor(Math.random()*couleurs.length)]}; z-index:9999; border-radius:${Math.random()>0.5?'50%':'2px'}; transition: transform 2s linear, top 2s linear, opacity 2s; pointer-events:none;`; document.body.appendChild(c); // Animation requestAnimationFrame(() => { c.style.top = '105vh'; c.style.transform = `rotate(${Math.random()*720}deg)`; c.style.opacity = '0'; }); // Nettoyage setTimeout(() => c.remove(), 2500); }, i * 30); } } // ========================================== // INITIALISATION ET REDIMENSIONNEMENT // ========================================== window.addEventListener('resize', () => { largeurCanvas = zoneJeu.clientWidth; hauteurCanvas = zoneJeu.clientHeight; canvas.width = largeurCanvas; canvas.height = hauteurCanvas; dessiner(); }); // Démarrer le dessin au chargement window.onload = () => { // Ajout du support pour les anciens navigateurs if (!CanvasRenderingContext2D.prototype.roundRect) { CanvasRenderingContext2D.prototype.roundRect = function(x, y, w, h, r) { if (w < 2 * r) r = w / 2; if (h < 2 * r) r = h / 2; this.beginPath(); this.moveTo(x + r, y); this.arcTo(x + w, y, x + w, y + h, r); this.arcTo(x + w, y + h, x, y + h, r); this.arcTo(x, y + h, x, y, r); this.arcTo(x, y, x + w, y, r); this.closePath(); return this; }; } dessiner(); };
